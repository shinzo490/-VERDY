'============================================================
' 条件付き書式 一覧出力マクロ （修正版）
' 対象環境 : Windows 10 / Excel 2019 （32bit・64bit 共通）
'
' 【概要】
'   選択したブックの全ワークシートから条件付き書式を読み取り、
'   このマクロブック（ThisWorkbook）の「CF_管理」シートに一覧出力します。
'   H 列には、その条件が成立したときの書式を静的に再現したプレビューを作ります。
'
' 【貼り付け方】
'   VBE の [挿入] - [標準モジュール] で新しいモジュールを作り、
'   このファイルの内容をすべて貼り付けてください。
'   ※ 他のマクロと同じモジュールには貼り付けないでください
'      （Private の共通関数名が重複するとコンパイルエラーになります）。
'
' 【参照設定】
'   不要です。Dictionary と FileDialog は遅延バインディングで呼び出しています。
'
' 【実行】
'   CreateConditionalFormatReport
'============================================================


Private Const OUTPUT_SHEET_NAME As String = "CF_管理"
Private Const COL_COUNT         As Long = 8      ' A〜H の 8 列


'============================================================
' メイン処理
'============================================================
Public Sub CreateConditionalFormatReport()

    Dim wbTarget As Workbook
    Dim wsTarget As Worksheet
    Dim wsOutput As Worksheet
    Dim rules As Collection
    Dim fc As Object
    Dim openedHere As Boolean

    Dim oldScreen As Boolean
    Dim oldEvents As Boolean
    Dim oldAlerts As Boolean
    Dim stateSaved As Boolean

    On Error GoTo ErrHandler

    Set wbTarget = SelectAndOpenWorkbook(openedHere)
    If wbTarget Is Nothing Then Exit Sub

    oldScreen = Application.ScreenUpdating
    oldEvents = Application.EnableEvents
    oldAlerts = Application.DisplayAlerts
    stateSaved = True

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    Set rules = New Collection

    For Each wsTarget In wbTarget.Worksheets

        ' 出力先シート自身は対象にしない（出力時にクリアされるため）
        If Not (wbTarget Is ThisWorkbook _
                And StrComp(wsTarget.Name, OUTPUT_SHEET_NAME, vbTextCompare) = 0) Then

            ' FormatConditions は Range のプロパティであり、Worksheet には存在しない。
            ' シート全体を表す Cells 経由で取得する。
            For Each fc In wsTarget.Cells.FormatConditions
                AddRule rules, wsTarget, fc
            Next fc

        End If
    Next wsTarget

    Set wsOutput = GetOrCreateOutputSheet(ThisWorkbook)
    wsOutput.Cells.Clear

    If rules.Count = 0 Then
        wsOutput.Range("A1").Value = "条件付き書式は見つかりませんでした。（対象 : " & wbTarget.Name & "）"
        wsOutput.Range("A1").Font.Bold = True
        MsgBox "条件付き書式は見つかりませんでした。" & vbCrLf & _
               "対象ブック : " & wbTarget.Name, vbInformation
        GoTo ExitHandler
    End If

    WriteReport wsOutput, rules
    FormatReportSheet wsOutput, rules.Count

    ' 出力結果を表示する（非表示シート等でも落ちないようにする）
    On Error Resume Next
    ThisWorkbook.Activate
    wsOutput.Activate
    On Error GoTo ErrHandler

    MsgBox "条件付き書式を " & rules.Count & " 件出力しました。" & vbCrLf & _
           "対象ブック : " & wbTarget.Name, vbInformation

ExitHandler:
    On Error Resume Next
    If openedHere And Not wbTarget Is Nothing Then wbTarget.Close SaveChanges:=False
    If stateSaved Then
        Application.DisplayAlerts = oldAlerts
        Application.EnableEvents = oldEvents
        Application.ScreenUpdating = oldScreen
    End If
    Set rules = Nothing
    Set wsOutput = Nothing
    Set wbTarget = Nothing
    On Error GoTo 0
    Exit Sub

ErrHandler:
    MsgBox "処理を中止しました。" & vbCrLf & vbCrLf & _
           "エラー番号 : " & Err.Number & vbCrLf & _
           "内容 : " & Err.Description, vbExclamation
    Resume ExitHandler

End Sub


'============================================================
' 1 ルール分の情報を集める
'   rules : Collection（要素は 1 ルール分の Dictionary）
'   キー  : SheetIndex / SheetName / Priority / RuleText /
'           FormatText / AppliesTo / StopIfTrue / TypeName / FC
'============================================================
Private Sub AddRule(ByVal rules As Collection, _
                    ByVal ws As Worksheet, _
                    ByVal fc As Object)

    Dim d As Object

    Set d = CreateObject("Scripting.Dictionary")
    d.CompareMode = vbTextCompare

    d.Add "SheetIndex", ws.Index
    d.Add "SheetName", ws.Name
    d.Add "Priority", GetPriority(fc)
    d.Add "RuleText", BuildRuleText(fc)
    d.Add "FormatText", BuildFormatText(fc)
    d.Add "AppliesTo", GetAppliesTo(fc)
    d.Add "StopIfTrue", GetStopIfTrue(fc)
    d.Add "TypeName", GetCFTypeName(fc)
    Set d("FC") = fc                      ' プレビュー用。値のコピーではなく参照。

    rules.Add d

End Sub


'============================================================
' 一覧の書き込み
'============================================================
Private Sub WriteReport(ByVal wsOutput As Worksheet, ByVal rules As Collection)

    Dim headers As Variant
    Dim arr As Variant
    Dim order() As Long
    Dim d As Object
    Dim i As Long
    Dim n As Long

    n = rules.Count

    headers = Array("シート", "優先順位", "条件付き書式ルール", "書式", _
                    "適用先", "真の場合は停止", "種類", "書式プレビュー")
    wsOutput.Range("A1").Resize(1, COL_COUNT).Value = headers

    SortRules rules, order

    ReDim arr(1 To n, 1 To COL_COUNT - 1)
    For i = 1 To n
        Set d = rules(order(i))
        arr(i, 1) = d("SheetName")
        arr(i, 2) = d("Priority")
        arr(i, 3) = d("RuleText")
        arr(i, 4) = d("FormatText")
        arr(i, 5) = d("AppliesTo")
        arr(i, 6) = d("StopIfTrue")
        arr(i, 7) = d("TypeName")
    Next i

    wsOutput.Range("A2").Resize(n, COL_COUNT - 1).Value = arr

    ' 並べ替え後の順番どおりに H 列のプレビューを作る
    For i = 1 To n
        Set d = rules(order(i))
        ApplyPreview wsOutput.Cells(i + 1, COL_COUNT), d("FC")
    Next i

End Sub


'============================================================
' 並べ替え（シートの並び順 → 優先順位 の昇順）
'   Priority はシート内で一意だが、重複していても止まらないよう
'   Dictionary のキーではなく添字の並べ替えで処理する。
'============================================================
Private Sub SortRules(ByVal rules As Collection, ByRef order() As Long)

    Dim keys As Variant
    Dim i As Long
    Dim n As Long

    n = rules.Count
    ReDim order(1 To n)
    ReDim keys(1 To n)

    For i = 1 To n
        order(i) = i
        keys(i) = Format$(rules(i)("SheetIndex"), "0000") & "|" & _
                  Format$(rules(i)("Priority"), "000000")
    Next i

    If n > 1 Then QuickSortPair keys, order, 1, n

End Sub


Private Sub QuickSortPair(ByRef keys As Variant, _
                          ByRef order() As Long, _
                          ByVal lo As Long, _
                          ByVal hi As Long)

    Dim i As Long
    Dim j As Long
    Dim pivot As String
    Dim tmpKey As Variant
    Dim tmpIdx As Long

    If lo >= hi Then Exit Sub

    i = lo
    j = hi
    pivot = CStr(keys((lo + hi) \ 2))

    Do While i <= j
        Do While StrComp(CStr(keys(i)), pivot, vbBinaryCompare) < 0
            i = i + 1
        Loop
        Do While StrComp(CStr(keys(j)), pivot, vbBinaryCompare) > 0
            j = j - 1
        Loop
        If i <= j Then
            tmpKey = keys(i): keys(i) = keys(j): keys(j) = tmpKey
            tmpIdx = order(i): order(i) = order(j): order(j) = tmpIdx
            i = i + 1
            j = j - 1
        End If
    Loop

    If lo < j Then QuickSortPair keys, order, lo, j
    If i < hi Then QuickSortPair keys, order, i, hi

End Sub


'============================================================
' ルール内容の文字列化
'============================================================
Private Function BuildRuleText(ByVal fc As Object) As String

    Dim f1 As String
    Dim f2 As String
    Dim txt As String

    Select Case GetCFType(fc)

        Case xlCellValue
            ' セル値条件だけは、表示用に先頭の = を取り除く
            f1 = RemoveLeadingEqual(GetFormula1(fc))
            f2 = RemoveLeadingEqual(GetFormula2(fc))

            Select Case GetOperator(fc)
                Case xlGreaterEqual: BuildRuleText = f1 & " 以上"
                Case xlLessEqual:    BuildRuleText = f1 & " 以下"
                Case xlLess:         BuildRuleText = f1 & " 未満"
                Case xlGreater:      BuildRuleText = f1 & " より大きい"
                Case xlEqual:        BuildRuleText = f1 & " に等しい"
                Case xlNotEqual:     BuildRuleText = f1 & " に等しくない"
                Case xlBetween:      BuildRuleText = f1 & " 以上 〜 " & f2 & " 以下"
                Case xlNotBetween:   BuildRuleText = f1 & " 未満 または " & f2 & " より大きい"
                Case Else:           BuildRuleText = "対象外の演算子（" & GetOperator(fc) & "）"
            End Select

        Case xlExpression
            ' 数式条件の = は数式そのものなので残す
            BuildRuleText = "数式 : " & GetFormula1(fc)

        Case xlTextString
            txt = GetTextString(fc)
            If Len(txt) = 0 Then txt = GetFormula1(fc)

            Select Case GetTextOperator(fc)
                Case xlContains:       BuildRuleText = "「" & txt & "」を含む"
                Case xlDoesNotContain: BuildRuleText = "「" & txt & "」を含まない"
                Case xlBeginsWith:     BuildRuleText = "「" & txt & "」で始まる"
                Case xlEndsWith:       BuildRuleText = "「" & txt & "」で終わる"
                Case Else:             BuildRuleText = "文字列 : " & txt
            End Select

        Case xlTimePeriod
            BuildRuleText = "日付 : " & GetTimePeriodText(fc)

        Case xlBlanksCondition:       BuildRuleText = "空白"
        Case xlNoBlanksCondition:     BuildRuleText = "空白ではない"
        Case xlErrorsCondition:       BuildRuleText = "エラー"
        Case xlNoErrorsCondition:     BuildRuleText = "エラーではない"
        Case xlUniqueValues:          BuildRuleText = GetDuplicateText(fc)
        Case xlTop10:                 BuildRuleText = GetTop10Text(fc)
        Case xlAboveAverageCondition: BuildRuleText = GetAboveAverageText(fc)
        Case xlDatabar:               BuildRuleText = "データバー"
        Case xlColorScale:            BuildRuleText = "カラースケール"
        Case xlIconSets:              BuildRuleText = "アイコンセット"
        Case Else:                    BuildRuleText = GetCFTypeName(fc) & "（詳細取得対象外）"

    End Select

End Function


Private Function RemoveLeadingEqual(ByVal formulaText As String) As String
    If Left$(formulaText, 1) = "=" Then
        RemoveLeadingEqual = Mid$(formulaText, 2)
    Else
        RemoveLeadingEqual = formulaText
    End If
End Function


'============================================================
' 書式内容の文字列化
'   条件付き書式は「指定していない属性」については Null を返す。
'   IsEmpty だけでは Null を弾けず、CStr(Null) で実行時エラー 94 になるため、
'   IsUsableValue（Empty / Null / エラー値をまとめて除外）で判定する。
'============================================================
Private Function BuildFormatText(ByVal fc As Object) As String

    Dim s As String
    Dim v As Variant

    Select Case GetCFType(fc)
        Case xlDatabar
            BuildFormatText = "データバー" & GetDataBarDetail(fc)
            Exit Function
        Case xlColorScale
            BuildFormatText = "カラースケール" & GetColorScaleDetail(fc)
            Exit Function
        Case xlIconSets
            BuildFormatText = "アイコンセット" & GetIconSetDetail(fc)
            Exit Function
    End Select

    v = TryGetFontColor(fc)
    If IsUsableValue(v) Then s = s & "文字色: " & GetColorName(v) & " "

    If IsTrueValue(TryGetFontBold(fc)) Then s = s & "太字 "
    If IsTrueValue(TryGetFontItalic(fc)) Then s = s & "斜体 "

    v = TryGetFontUnderline(fc)
    If IsUsableValue(v) Then
        If CLng(v) <> xlUnderlineStyleNone Then s = s & "下線 "
    End If

    If IsTrueValue(TryGetFontStrikethrough(fc)) Then s = s & "取り消し線 "

    v = TryGetFillColor(fc)
    If IsUsableValue(v) Then s = s & "塗り: " & GetColorName(v) & " "

    v = TryGetNumberFormat(fc)
    If IsUsableValue(v) Then
        If Len(CStr(v)) > 0 And CStr(v) <> "General" Then
            s = s & "表示形式: " & CStr(v) & " "
        End If
    End If

    If Len(Trim$(s)) = 0 Then s = "（書式の指定なし）"
    BuildFormatText = Trim$(s)

End Function


'============================================================
' 書式プレビュー
'   条件そのものは複製せず、条件が成立したときの書式だけを静的に再現する。
'   属性ごとに Null 判定と Err.Clear を行うため、一部の属性が未指定でも
'   設定できた分はそのまま反映される。
'============================================================
Private Sub ApplyPreview(ByVal previewCell As Range, ByVal fc As Object)

    Dim v As Variant
    Dim applied As Boolean

    On Error Resume Next

    previewCell.ClearFormats
    previewCell.Value = "Aa 123"
    previewCell.HorizontalAlignment = xlCenter
    Err.Clear

    v = TryGetFontColor(fc)
    If IsUsableValue(v) Then
        previewCell.Font.Color = CLng(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetFillColor(fc)
    If IsUsableValue(v) Then
        previewCell.Interior.Color = CLng(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetFontBold(fc)
    If IsUsableValue(v) Then
        previewCell.Font.Bold = CBool(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetFontItalic(fc)
    If IsUsableValue(v) Then
        previewCell.Font.Italic = CBool(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetFontUnderline(fc)
    If IsUsableValue(v) Then
        previewCell.Font.Underline = CLng(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetFontStrikethrough(fc)
    If IsUsableValue(v) Then
        previewCell.Font.Strikethrough = CBool(v)
        If Err.Number = 0 Then applied = True
        Err.Clear
    End If

    v = TryGetNumberFormat(fc)
    If IsUsableValue(v) Then
        If Len(CStr(v)) > 0 Then
            previewCell.NumberFormat = CStr(v)
            If Err.Number = 0 Then applied = True
        End If
        Err.Clear
    End If

    If Not applied Then
        ' データバー・カラースケール・アイコンセットなど、
        ' セル書式として再現できないものはここに入る
        previewCell.Value = "（プレビュー対象外）"
        previewCell.HorizontalAlignment = xlCenter
    End If

    Err.Clear
    On Error GoTo 0

End Sub

Private Function GetTop10Text(ByVal fc As Object) As String
    Dim label As String
    Dim unitText As String

    On Error GoTo Failed

    ' 上位／下位の区分は TopBottom プロパティ（xlTop10Top / xlTop10Bottom）で判定する。
    ' "Top10" というプロパティは存在しない。
    If fc.TopBottom = xlTop10Top Then label = "上位 " Else label = "下位 "
    If fc.Percent Then unitText = "%" Else unitText = " 項目"

    GetTop10Text = label & CStr(fc.Rank) & unitText
    Exit Function
Failed:
    GetTop10Text = "上位／下位"
End Function


Private Function GetAboveAverageText(ByVal fc As Object) As String
    On Error GoTo Failed

    ' XlAboveBelow の値をハードコードすると取り違えるおそれがあるため、
    ' 定数名で判定する。標準偏差の倍数は NumStDev から取得する。
    Select Case fc.AboveBelow
        Case xlAboveAverage:      GetAboveAverageText = "平均より上"
        Case xlBelowAverage:      GetAboveAverageText = "平均より下"
        Case xlEqualAboveAverage: GetAboveAverageText = "平均以上"
        Case xlEqualBelowAverage: GetAboveAverageText = "平均以下"
        Case xlAboveStdDev:       GetAboveAverageText = "平均より " & GetNumStDev(fc) & " 標準偏差上"
        Case xlBelowStdDev:       GetAboveAverageText = "平均より " & GetNumStDev(fc) & " 標準偏差下"
        Case Else:                GetAboveAverageText = "平均条件"
    End Select
    Exit Function
Failed:
    GetAboveAverageText = "平均条件"
End Function

'--- Empty / Null / エラー値をまとめて除外する -------------
Private Function IsUsableValue(ByVal v As Variant) As Boolean
    If IsEmpty(v) Then Exit Function
    If IsNull(v) Then Exit Function
    If IsError(v) Then Exit Function
    IsUsableValue = True
End Function


Private Function IsTrueValue(ByVal v As Variant) As Boolean
    If Not IsUsableValue(v) Then Exit Function

    On Error Resume Next
    IsTrueValue = CBool(v)
    If Err.Number <> 0 Then
        IsTrueValue = False
        Err.Clear
    End If
    On Error GoTo 0
End Function


'--- 色名（既知の色以外は RGB 値で表示）-------------------
Private Function GetColorName(ByVal colorValue As Variant) As String
    Dim clr As Long
    Dim r As Long
    Dim g As Long
    Dim b As Long

    On Error GoTo Failed

    ' テーマ色などで負の値が返っても壊れないよう、下位 24 ビットだけを使う
    clr = CLng(colorValue) And &HFFFFFF

    Select Case clr
        Case RGB(255, 0, 0):     GetColorName = "赤"
        Case RGB(0, 0, 255):     GetColorName = "青"
        Case RGB(0, 128, 0):     GetColorName = "緑"
        Case RGB(255, 255, 0):   GetColorName = "黄"
        Case RGB(255, 255, 255): GetColorName = "白"
        Case RGB(0, 0, 0):       GetColorName = "黒"
        Case RGB(255, 192, 0):   GetColorName = "ゴールド"
        Case RGB(255, 165, 0):   GetColorName = "オレンジ"
        Case RGB(192, 192, 192): GetColorName = "グレー"
        Case RGB(255, 199, 206): GetColorName = "薄い赤"
        Case RGB(198, 239, 206): GetColorName = "薄い緑"
        Case RGB(255, 235, 156): GetColorName = "薄い黄"
        Case RGB(156, 0, 6):     GetColorName = "濃い赤"
        Case RGB(0, 97, 0):      GetColorName = "濃い緑"
        Case RGB(156, 101, 0):   GetColorName = "濃い黄"
        Case Else
            r = clr And &HFF&
            g = (clr \ &H100&) And &HFF&
            b = (clr \ &H10000) And &HFF&
            GetColorName = "RGB(" & r & "," & g & "," & b & ")"
    End Select
    Exit Function

Failed:
    GetColorName = "（取得不可）"
End Function


'============================================================
' 対象ブックの選択（既に開いていればそれを使う）
'============================================================
Private Function SelectAndOpenWorkbook(ByRef openedHere As Boolean) As Workbook

    Dim fd As Object
    Dim selectedPath As String
    Dim wb As Workbook

    openedHere = False

    ' 3 = msoFileDialogFilePicker
    ' 定数名を使うと Office ライブラリへの参照設定が必要になるため、数値で指定する。
    Set fd = Application.FileDialog(3)
    With fd
        .Title = "条件付き書式を取得する Excel ファイルを選択してください"
        .AllowMultiSelect = False
        .Filters.Clear
        .Filters.Add "Excel ファイル", "*.xlsx;*.xlsm;*.xlsb;*.xls"

        If .Show <> -1 Then Exit Function
        selectedPath = .SelectedItems(1)
    End With

    For Each wb In Application.Workbooks
        If StrComp(wb.FullName, selectedPath, vbTextCompare) = 0 Then
            Set SelectAndOpenWorkbook = wb
            Exit Function
        End If
    Next wb

    ' Open に失敗したときに openedHere が True のまま残ると、
    ' 終了処理で Nothing.Close を呼んでしまうため、成功後に True にする。
    Set SelectAndOpenWorkbook = Workbooks.Open(Filename:=selectedPath, ReadOnly:=True, UpdateLinks:=0)
    openedHere = True

End Function


'============================================================
' 出力先シートの取得（無ければ作成）
'============================================================
Private Function GetOrCreateOutputSheet(ByVal wb As Workbook) As Worksheet

    Dim ws As Worksheet
    Dim sh As Object

    For Each ws In wb.Worksheets
        If StrComp(ws.Name, OUTPUT_SHEET_NAME, vbTextCompare) = 0 Then
            Set GetOrCreateOutputSheet = ws
            Exit Function
        End If
    Next ws

    ' 同名のグラフシート等があると Add 後の改名で失敗するため、先に確認する
    For Each sh In wb.Sheets
        If StrComp(sh.Name, OUTPUT_SHEET_NAME, vbTextCompare) = 0 Then
            Err.Raise vbObjectError + 1002, "GetOrCreateOutputSheet", _
                      "「" & OUTPUT_SHEET_NAME & "」という名前のシートが、" & _
                      "ワークシート以外の種類で存在します。"
        End If
    Next sh

    Set GetOrCreateOutputSheet = wb.Worksheets.Add(After:=wb.Sheets(wb.Sheets.Count))
    GetOrCreateOutputSheet.Name = OUTPUT_SHEET_NAME

End Function


'============================================================
' 出力シートの体裁
'============================================================
Private Sub FormatReportSheet(ByVal ws As Worksheet, ByVal ruleCount As Long)

    Dim lastRow As Long

    lastRow = ruleCount + 1

    With ws
        With .Range("A1:H1")
            .Font.Bold = True
            .Font.Color = RGB(255, 255, 255)
            .Interior.Color = RGB(31, 78, 121)
            .HorizontalAlignment = xlCenter
        End With

        With .Range("A1:H" & lastRow)
            .Borders.LineStyle = xlContinuous
            .VerticalAlignment = xlCenter
        End With

        .Columns("A").ColumnWidth = 18
        .Columns("B").ColumnWidth = 10
        .Columns("C").ColumnWidth = 32
        .Columns("D").ColumnWidth = 34
        .Columns("E").ColumnWidth = 28
        .Columns("F").ColumnWidth = 15
        .Columns("G").ColumnWidth = 16
        .Columns("H").ColumnWidth = 16

        .Range("C2:E" & lastRow).WrapText = True

        ' Range.AutoFilter は引数なしで呼ぶとトグル動作になる。
        ' Cells.Clear をしても AutoFilterMode は True のまま残るため、
        ' 2 回目以降の実行で逆に解除されてしまう。先に一度解除してからオンにする。
        If .AutoFilterMode Then .AutoFilterMode = False
        .Range("A1:H" & lastRow).AutoFilter

        .Rows("1:" & lastRow).EntireRow.AutoFit
    End With

End Sub
